viewof selected_flag = Inputs.select(flag_files, {
  label: "Choose a flag:",
  value: "us.png"
})

// Choose what "close" means
viewof metric = Inputs.radio(
  ["Overall Color", "Color + Pattern", "Shape (Aspect Ratio)"],
  { label: "Closeness by:", value: "Overall Color" }
)

// Choose closest or most distant matches
viewof match_type = Inputs.radio(
  new Map([
    ["3 Closest Matches", "closest"],
    ["3 Most Distant Matches", "distant"]
  ]),
  { label: "Display:", value: "closest" }
)

// Display the selected flag image
html`<img src="flags_files/w2560/${selected_flag}" style="max-width: 400px; border: 1px solid #ccc; margin: 1rem 0;">`
chosen_matches = matches[metric][selected_flag][match_type]

html`<div style="display: flex; gap: 1.5rem; flex-wrap: wrap; margin-top: 0.5rem;">
  ${chosen_matches.map(f => html`<figure style="margin: 0; text-align: center;">
    <img src="flags_files/w2560/${f}" style="height: 120px; border: 1px solid #ccc;">
    <figcaption style="margin-top: 0.25rem; font-family: monospace;">${f}</figcaption>
  </figure>`)}
</div>`
